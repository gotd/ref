// waitReady blocks until ready is closed or ctx is done and reports whether
// ready was closed.
//
// Readiness wins when both happen: a request may complete on the fresh
// connection and cancel ctx before this goroutine is scheduled, and a plain
// select would then drop the ready event at random.
func ( context.Context,  <-chan struct{}) bool {
	select {
	case <-:
		return true
	case <-.Done():
		select {
		case <-:
			return true
		default:
			return false
		}
	}
}

func ( *Client) ( error) bool {
	// See https://github.com/gotd/td/issues/1458.
	if errors.Is(, exchange.ErrKeyFingerprintNotFound) {
		return true
	}
	if tgerr.Is(, "AUTH_KEY_UNREGISTERED", "SESSION_EXPIRED", "AUTH_KEY_DUPLICATED") {
		return true
	}
	if tgerr.Is(, "AUTH_KEY_PERM_EMPTY") {
		// PFS: the temporary key binding was lost, a reconnect re-binds.
		return false
	}
	if auth.IsUnauthorized() {
		return true
	}
	return false
}

// Run starts client session and blocks until connection close.
// The f callback is called on successful session initialization and Run
// will return on f() result.
//
// Context of callback will be canceled if fatal error is detected.
// The ctx is used for background operations like updates handling or pools.
//
// Run is single-use: create a new Client to run again. See Start, Close and
// Wait for the classic approach without explicit callback; Run and Start are
// mutually exclusive on one Client (ErrAlreadyStarted). Done and Wait report
// the result of Run once it returned.
func ( *Client) ( context.Context,  func( context.Context) error) ( error) {
	if !.life.mode.CompareAndSwap(int32(modeIdle), int32(modeRun)) {
		switch lifecycleMode(.life.mode.Load()) {
		case modeStart:
			return errDrivenByStart
		case modeClosed:
			return ErrClientClosed
		}
		// Run again, fails as the client is closed.
		return .run(, )
	}
	// First Run: report its result via Done and Wait.
	defer func() { .finishStart(, false) }()
	return .run(, )
}
