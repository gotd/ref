// sink tracks download progress.
type sink struct {
	done     int64
	total    int64
	progress ProgressFunc
}

func ( *sink) ( int) {
	.done += int64()
	if .progress != nil {
		.progress(.done, .total)
	}
}

func ( io.WriterAt,  sink,  <-chan block) func(context.Context) error {

}
