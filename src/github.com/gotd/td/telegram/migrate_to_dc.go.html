// ensureRestart restarts the primary connection and waits until it is ready.
// Reason is restartTimeout or restartMigration, for metrics and logs.
func ( *Client) ( context.Context,  string) error {
	.log.Debug(, "Triggering restart", log.String("reason", ))
	.resetReady()

	select {
	case <-.Done():
		return .Err()
	case .restart <- struct{}{}:
		// Counted once sent, not when abandoned on cancel.
		.meters().restarts.Add(, 1, metric.WithAttributes(attribute.String("reason", )))
		.log.Debug(, "Restart initialized")
	}

	select {
	case <-.Done():
		return .Err()
	case <-.ready.Ready():
		.log.Info(, "Restart ensured")
		return nil
	}
}
